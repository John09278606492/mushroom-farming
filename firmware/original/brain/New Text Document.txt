String getFieldValue(String data, String key) {
  int start = data.indexOf(key + "=");
  if (start < 0) return "";
  start += key.length() + 1;
  int end = data.indexOf(',', start);
  if (end < 0) end = data.length();
  return data.substring(start, end);
}

void sendCommand(const String& cmd) {
  slaveSerial.println(cmd);
  Serial.print("SENT: ");
  Serial.println(cmd);
}

void sendRTCToBody(int year, int month, int day, int hour, int minute, int second) {
  char buf[32];
  snprintf(buf, sizeof(buf), "SET_RTC=%04d-%02d-%02d %02d:%02d:%02d",
           year, month, day, hour, minute, second);
  sendCommand(String(buf));
}

void sendSprinklerSchedule(int idx, bool en, int hour, int minute, int dur, String daysDigits) {
  String cmd = "SET_SPRINKLER_SCHED=" +
               String(idx) + "," +
               String(en ? 1 : 0) + "," +
               String(hour) + "," +
               String(minute) + "," +
               String(dur) + "," +
               daysDigits;
  sendCommand(cmd);
}

void sendLightSchedule(int idx, bool en, int hour, int minute, int dur, String daysDigits) {
  String cmd = "SET_LIGHT_SCHED=" +
               String(idx) + "," +
               String(en ? 1 : 0) + "," +
               String(hour) + "," +
               String(minute) + "," +
               String(dur) + "," +
               daysDigits;
  sendCommand(cmd);
}

int getRollerInt(lv_obj_t *roller) {
  char buf[16];
  lv_roller_get_selected_str(roller, buf, sizeof(buf));
  return atoi(buf);
}

String getCheckedDays(lv_obj_t *checks[7]) {
  String result = "";
  for (int i = 0; i < 7; i++) {
    if (lv_obj_has_state(checks[i], LV_STATE_CHECKED)) {
      result += String(i);
    }
  }
  return result;
}

void fillNumericOptions(char *buf, size_t size, int start, int end, bool fourDigits = false) {
  buf[0] = '\0';
  for (int i = start; i <= end; i++) {
    char line[16];
    if (fourDigits) snprintf(line, sizeof(line), "%04d", i);
    else snprintf(line, sizeof(line), "%02d", i);
    strncat(buf, line, size - strlen(buf) - 1);
    if (i < end) strncat(buf, "\n", size - strlen(buf) - 1);
  }
}

// =====================================================
// LVGL DISPLAY / TOUCH CALLBACKS
// =====================================================
void my_disp_flush(lv_display_t *display, const lv_area_t *area, uint8_t *px_map) {
  uint32_t w = (area->x2 - area->x1 + 1);
  uint32_t h = (area->y2 - area->y1 + 1);
  gfx->draw16bitRGBBitmap(area->x1, area->y1, (uint16_t *)px_map, w, h);
  lv_display_flush_ready(display);
}

void my_touch_read(lv_indev_t *indev_drv, lv_indev_data_t *data) {
  ts.read();
  if (ts.isTouched) {
    data->state   = LV_INDEV_STATE_PRESSED;
    data->point.x = ts.points[0].x;
    data->point.y = ts.points[0].y;
  } else {
    data->state = LV_INDEV_STATE_RELEASED;
  }
}

// =====================================================
// REFRESH UI
// =====================================================
void refreshUI() {
  if (lblTopTime) {
    lv_label_set_text(lblTopTime, ("Time: " + rtcTime).c_str());
  }

  if (lblTopEnv) {
    String env = "Humidity: " + String(humidity, 1) +
                 "%   Temp: " + String(temperature, 1) +
                 " C   CO2: " + String(co2ppm) + " ppm";
    lv_label_set_text(lblTopEnv, env.c_str());
  }

  if (chipSprLabel) {
    lv_label_set_text(chipSprLabel, sprinklerState ? "ON" : "OFF");
    lv_obj_set_style_text_color(chipSprLabel, sprinklerState ? lv_palette_main(LV_PALETTE_GREEN) : lv_palette_main(LV_PALETTE_RED), 0);
  }

  if (chipLightLabel) {
    lv_label_set_text(chipLightLabel, lightState ? "ON" : "OFF");
    lv_obj_set_style_text_color(chipLightLabel, lightState ? lv_palette_main(LV_PALETTE_GREEN) : lv_palette_main(LV_PALETTE_RED), 0);
  }

  if (chipAutoLabel) {
    lv_label_set_text(chipAutoLabel, autoMode ? "AUTO" : "MANUAL");
    lv_obj_set_style_text_color(chipAutoLabel, autoMode ? lv_palette_main(LV_PALETTE_BLUE) : lv_palette_main(LV_PALETTE_ORANGE), 0);
  }

  if (chipWaterLabel) {
    lv_label_set_text(chipWaterLabel, lowWater ? "LOW" : "OK");
    lv_obj_set_style_text_color(chipWaterLabel, lowWater ? lv_palette_main(LV_PALETTE_ORANGE) : lv_palette_main(LV_PALETTE_GREEN), 0);
  }

  if (lblSprStatus) {
    lv_label_set_text(lblSprStatus, sprinklerState ? "ON" : "OFF");
    lv_obj_set_style_text_color(lblSprStatus, sprinklerState ? lv_palette_main(LV_PALETTE_GREEN) : lv_palette_main(LV_PALETTE_RED), 0);
  }

  if (lblLightStatus) {
    lv_label_set_text(lblLightStatus, lightState ? "ON" : "OFF");
    lv_obj_set_style_text_color(lblLightStatus, lightState ? lv_palette_main(LV_PALETTE_GREEN) : lv_palette_main(LV_PALETTE_RED), 0);
  }

  if (lblSprSched) {
    lv_label_set_text(lblSprSched, ("Current: " + sprinklerSchedText).c_str());
  }

  if (lblLightSched) {
    lv_label_set_text(lblLightSched, ("Current: " + lightSchedText).c_str());
  }

  if (lblRTCStatus) {
    lv_label_set_text(lblRTCStatus, ("Current RTC: " + rtcTime).c_str());
  }
}

// =====================================================
// STATUS PARSE
// =====================================================
void parseStatus(String line) {
  if (!line.startsWith("STATUS,")) return;

  String v;
  v = getFieldValue(line, "AUTO");             if (v.length()) autoMode = v.toInt();
  v = getFieldValue(line, "H");                if (v.length()) humidity = v.toFloat();
  v = getFieldValue(line, "T");                if (v.length()) temperature = v.toFloat();
  v = getFieldValue(line, "CO2");              if (v.length()) co2ppm = v.toInt();
  v = getFieldValue(line, "LOWWATER");         if (v.length()) lowWater = v.toInt();
  v = getFieldValue(line, "MIST");             if (v.length()) mistState = v.toInt();
  v = getFieldValue(line, "REFILL");           if (v.length()) refillState = v.toInt();
  v = getFieldValue(line, "SPRINKLER");        if (v.length()) sprinklerState = v.toInt();
  v = getFieldValue(line, "FAN");              if (v.length()) fanState = v.toInt();
  v = getFieldValue(line, "LIGHT");            if (v.length()) lightState = v.toInt();
  v = getFieldValue(line, "TIME");             if (v.length()) rtcTime = v;
  v = getFieldValue(line, "SPRINKLER_SCHED");  if (v.length()) sprinklerSchedText = v;
  v = getFieldValue(line, "LIGHT_SCHED");      if (v.length()) lightSchedText = v;

  refreshUI();
}

// =====================================================
// INIT DISPLAY + TOUCH
// =====================================================
void initDisplayAndTouch() {
  pinMode(GFX_BL, OUTPUT);
  digitalWrite(GFX_BL, HIGH);

  gfx->begin();
  gfx->fillScreen(0x0000);

  Wire.begin(TOUCH_SDA, TOUCH_SCL);
  ts.begin();
  ts.setRotation(ROTATION_NORMAL);

  lv_init();

  disp = lv_display_create(SCREEN_W, SCREEN_H);
  lv_display_set_flush_cb(disp, my_disp_flush);
  lv_display_set_buffers(disp, buf1, NULL, sizeof(buf1), LV_DISPLAY_RENDER_MODE_PARTIAL);

  indev = lv_indev_create();
  lv_indev_set_type(indev, LV_INDEV_TYPE_POINTER);
  lv_indev_set_read_cb(indev, my_touch_read);
}

// =====================================================
// SETUP
// =====================================================
void setup() {
  Serial.begin(115200);

  slaveSerial.begin(115200, SERIAL_8N1, LCD_RX_FROM_SLAVE, LCD_TX_TO_SLAVE);

  initDisplayAndTouch();

  buildHomeScreen();
  buildSprinklerScreen();
  buildLightScreen();
  buildRTCScreen();

  lv_screen_load(screenHome);
  refreshUI();

  Serial.println("HMI UI V3 READY");
}

// =====================================================
// LOOP
// =====================================================
void loop() {
  while (slaveSerial.available()) {
    String line = slaveSerial.readStringUntil('\n');
    line.trim();
    parseStatus(line);
  }

  static unsigned long lastReq = 0;
  if (millis() - lastReq > 2000) {
    lastReq = millis();
    sendCommand("GET_STATUS");
  }

  lv_timer_handler();
  delay(5);
}
